    2. Create backend/app/core/security.py:
       - Import: jwt, datetime, timedelta, passlib.context.CryptContext, app.core.config.settings
       - pwd_ctx = CryptContext(schemes=["argon2"], deprecated="auto")
       - hash_password(plain: str) -> str: return pwd_ctx.hash(plain)
       - verify_password(plain: str, hashed: str) -> bool: return pwd_ctx.verify(plain, hashed)
       - create_access_token(data: dict, expires_minutes: int = 30) -> str:
           payload = data.copy()
           payload["exp"] = datetime.utcnow() + timedelta(minutes=expires_minutes)
           return jwt.encode(payload, settings.jwt_secret, algorithm="HS256")
       - decode_access_token(token: str) -> dict:
           return jwt.decode(token, settings.jwt_secret, algorithms=["HS256"])
           # raises jwt.PyJWTError on failure — callers catch it

    3. Add to backend/app/core/config.py (Settings class):
       - jwt_secret: str = Field(..., env="JWT_SECRET")
       - admin_username: str = Field("admin", env="ADMIN_USERNAME")
       - admin_password_hash: str = Field(..., env="ADMIN_PASSWORD_HASH")
         # This is the argon2 hash of the admin password — set at deploy time
         # Generate with: python -c "from passlib.context import CryptContext; print(CryptContext(schemes=['argon2']).hash('yourpassword'))"
       - frontend_url: str = Field("http://localhost:5173", env="FRONTEND_URL")
       - cookie_samesite: str = Field("lax", env="COOKIE_SAMESITE")
